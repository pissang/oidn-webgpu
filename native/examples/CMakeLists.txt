cmake_minimum_required(VERSION 3.20)
project(oidn_webgpu_c_examples LANGUAGES C)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
# Shipped Windows libraries use the DLL CRT, also for a Debug host build.
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
if(APPLE)
  set(CMAKE_INSTALL_RPATH "@loader_path/../lib")
elseif(UNIX)
  set(CMAKE_INSTALL_RPATH "$ORIGIN/../lib")
endif()

find_package(oidn_webgpu CONFIG REQUIRED)

add_executable(denoise denoise.c)
target_link_libraries(denoise PRIVATE oidn_webgpu::oidn_webgpu)
add_executable(plan plan.c)
target_link_libraries(plan PRIVATE oidn_webgpu::oidn_webgpu_core)
set(examples denoise plan)
set(shared_examples denoise)
if(APPLE)
  enable_language(OBJC CXX OBJCXX)
  add_executable(metal_gpu metal_gpu.m)
  # Older supported CMake releases omit Objective-C runtime RPATH flags.
  # The C driver links these ObjC objects too and preserves build/install RPATH.
  set_target_properties(metal_gpu PROPERTIES LINKER_LANGUAGE C)
  target_compile_options(metal_gpu PRIVATE -fobjc-arc)
  target_link_libraries(metal_gpu PRIVATE oidn_webgpu::oidn_webgpu "-framework Metal" "-framework Foundation")
  add_executable(metal_gpu_cpp metal_gpu_cpp.mm)
  set_target_properties(metal_gpu_cpp PROPERTIES LINKER_LANGUAGE CXX OBJCXX_STANDARD 17 OBJCXX_STANDARD_REQUIRED ON)
  target_compile_options(metal_gpu_cpp PRIVATE -fobjc-arc)
  target_link_libraries(metal_gpu_cpp PRIVATE oidn_webgpu::cpp "-framework Metal" "-framework Foundation")
  list(APPEND examples metal_gpu_cpp)
  list(APPEND shared_examples metal_gpu_cpp)
  list(APPEND examples metal_gpu)
  list(APPEND shared_examples metal_gpu)
endif()
if(WIN32)
  option(OIDNW_BUILD_VULKAN_GPU_EXAMPLE "Build Vulkan GPU DLL consumer" OFF)
  option(OIDNW_BUILD_DX12_GPU_EXAMPLE "Build DX12 GPU DLL consumer" OFF)
  if(OIDNW_BUILD_VULKAN_GPU_EXAMPLE OR OIDNW_BUILD_DX12_GPU_EXAMPLE)
    enable_language(CXX)
    install(DIRECTORY shaders/ DESTINATION bin/shaders)
  endif()
  if(OIDNW_BUILD_VULKAN_GPU_EXAMPLE)
    find_path(Vulkan_INCLUDE_DIR vulkan/vulkan.h HINTS "$ENV{VULKAN_SDK}/Include" REQUIRED)
    add_executable(vulkan_gpu vulkan_gpu.cpp)
    target_compile_features(vulkan_gpu PRIVATE cxx_std_17)
    target_include_directories(vulkan_gpu PRIVATE "${Vulkan_INCLUDE_DIR}")
    target_link_libraries(vulkan_gpu PRIVATE oidn_webgpu::cpp)
    list(APPEND examples vulkan_gpu)
    list(APPEND shared_examples vulkan_gpu)
  endif()
  if(OIDNW_BUILD_DX12_GPU_EXAMPLE)
    find_path(DX12_AGILITY_INCLUDE_DIR d3d12.h HINTS "$ENV{WGPU_DX12_AGILITY_SDK_PATH}/../../include" REQUIRED)
    add_executable(dx12_gpu dx12_gpu.cpp)
    target_compile_features(dx12_gpu PRIVATE cxx_std_17)
    target_include_directories(dx12_gpu PRIVATE "${DX12_AGILITY_INCLUDE_DIR}")
    target_link_libraries(dx12_gpu PRIVATE oidn_webgpu::cpp d3d12 dxgi d3dcompiler)
    list(APPEND examples dx12_gpu)
    list(APPEND shared_examples dx12_gpu)
  endif()
  add_executable(validate_windows validate_windows.c)
  target_link_libraries(validate_windows PRIVATE oidn_webgpu::oidn_webgpu)
  list(APPEND examples validate_windows)
  list(APPEND shared_examples validate_windows)
  foreach(example IN LISTS shared_examples)
    add_custom_command(TARGET ${example} POST_BUILD
      COMMAND "${CMAKE_COMMAND}" -E copy_if_different
        "$<TARGET_FILE:oidn_webgpu::oidn_webgpu>" "$<TARGET_FILE_DIR:${example}>"
      VERBATIM)
  endforeach()
  set(runtime_destination bin)
else()
  set(runtime_destination lib)
endif()

install(TARGETS ${examples} RUNTIME DESTINATION bin)
install(FILES "$<TARGET_FILE:oidn_webgpu::oidn_webgpu>" DESTINATION "${runtime_destination}")
