cmake_minimum_required(VERSION 3.20)
project(oidn_webgpu_cuda LANGUAGES CXX)
option(OIDNW_BUILD_CUDA_ADAPTER "Build the optional Windows CUDA/DX12 adapter" OFF)
option(OIDNW_CUDA_BUILD_TESTS "Build source-only CUDA adapter fault checks" OFF)
if(NOT OIDNW_BUILD_CUDA_ADAPTER)
  return()
endif()
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "The CUDA/DX12 adapter requires 64-bit Windows")
endif()
find_package(oidn_webgpu CONFIG REQUIRED)
find_package(CUDAToolkit 12.5 REQUIRED)
find_path(DX12_AGILITY_INCLUDE_DIR d3d12.h HINTS "$ENV{WGPU_DX12_AGILITY_SDK_PATH}/../../include" REQUIRED)
set(OIDNW_CUDA_PUBLIC_INCLUDE "${CMAKE_CURRENT_SOURCE_DIR}/../include")
if(NOT EXISTS "${OIDNW_CUDA_PUBLIC_INCLUDE}/oidn_webgpu_cuda.hpp")
  set(OIDNW_CUDA_PUBLIC_INCLUDE "${CMAKE_CURRENT_SOURCE_DIR}/../../include")
endif()
add_library(oidn_webgpu_cuda STATIC adapter.cpp)
add_library(oidn_webgpu::cuda ALIAS oidn_webgpu_cuda)
set_target_properties(oidn_webgpu_cuda PROPERTIES MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
target_compile_features(oidn_webgpu_cuda PUBLIC cxx_std_17)
target_compile_options(oidn_webgpu_cuda PRIVATE /EHsc)
target_include_directories(oidn_webgpu_cuda PUBLIC
  $<BUILD_INTERFACE:${OIDNW_CUDA_PUBLIC_INCLUDE}> $<INSTALL_INTERFACE:include>)
target_include_directories(oidn_webgpu_cuda PRIVATE "${DX12_AGILITY_INCLUDE_DIR}")
target_link_libraries(oidn_webgpu_cuda PUBLIC oidn_webgpu::cpp CUDA::cudart CUDA::cuda_driver PRIVATE d3d12 dxgi)
install(TARGETS oidn_webgpu_cuda EXPORT oidn_webgpu_cudaTargets ARCHIVE DESTINATION lib)
install(FILES "${OIDNW_CUDA_PUBLIC_INCLUDE}/oidn_webgpu_cuda.hpp" DESTINATION include)
set_target_properties(oidn_webgpu_cuda PROPERTIES EXPORT_NAME cuda)
install(EXPORT oidn_webgpu_cudaTargets NAMESPACE oidn_webgpu:: DESTINATION lib/cmake/oidn_webgpu_cuda)
install(FILES oidn_webgpu_cudaConfig.cmake DESTINATION lib/cmake/oidn_webgpu_cuda)
if(OIDNW_CUDA_BUILD_TESTS)
  if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../tests/cuda_adapter.cpp")
    message(FATAL_ERROR "OIDNW_CUDA_BUILD_TESTS requires the repository checkout")
  endif()
  add_library(oidn_webgpu_cuda_faults STATIC adapter.cpp)
  target_compile_definitions(oidn_webgpu_cuda_faults PRIVATE oidnw_gpu_submit=oidnw_cuda_test_submit
    oidnw_gpu_poll=oidnw_cuda_test_poll cuDeviceGetLuid=oidnw_cuda_test_luid)
  target_compile_features(oidn_webgpu_cuda_faults PRIVATE cxx_std_17)
  target_compile_options(oidn_webgpu_cuda_faults PRIVATE /EHsc)
  set_target_properties(oidn_webgpu_cuda_faults PROPERTIES MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
  target_include_directories(oidn_webgpu_cuda_faults PRIVATE "${OIDNW_CUDA_PUBLIC_INCLUDE}" "${DX12_AGILITY_INCLUDE_DIR}")
  target_link_libraries(oidn_webgpu_cuda_faults PUBLIC oidn_webgpu::cpp CUDA::cudart CUDA::cuda_driver d3d12 dxgi)
  add_executable(cuda_adapter_failures ../tests/cuda_adapter.cpp)
  target_compile_features(cuda_adapter_failures PRIVATE cxx_std_17)
  target_compile_options(cuda_adapter_failures PRIVATE /EHsc)
  target_include_directories(cuda_adapter_failures PRIVATE "${OIDNW_CUDA_PUBLIC_INCLUDE}")
  set_target_properties(cuda_adapter_failures PROPERTIES MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
  target_link_libraries(cuda_adapter_failures PRIVATE oidn_webgpu_cuda_faults)
  add_custom_command(TARGET cuda_adapter_failures POST_BUILD COMMAND "${CMAKE_COMMAND}" -E copy_if_different
    "$<TARGET_FILE:oidn_webgpu::oidn_webgpu>" "$<TARGET_FILE_DIR:cuda_adapter_failures>")
endif()
